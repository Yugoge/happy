    33	function unwrapShellCommand(command: string): string {
    34	    const match = command.match(/^(?:\/bin\/)?(?:ba|z)?sh\s+-l?c\s+([\s\S]+)$/);
    35	    if (!match) return command;
    36	    const inner = match[1].trim();
    37	    if (
    38	        (inner.startsWith('"') && inner.endsWith('"')) ||
    39	        (inner.startsWith("'") && inner.endsWith("'"))
    40	    ) {
    41	        return inner.slice(1, -1).trim();
    42	    }
    43	    return inner;
    44	}

    46	function getCommandCandidates(input: ToolCall['input']): string[] {
    47	    const candidates = new Set<string>();
    48	    const add = (value: unknown) => {
    49	        if (typeof value !== 'string') return;
    50	        const trimmed = value.trim();
    51	        if (!trimmed) return;
    52	        candidates.add(trimmed);
    53	        candidates.add(unwrapShellCommand(trimmed));
    54	    };
    55	    add(stringifyToolCommand(input?.command));
    56	    if (Array.isArray(input?.command)) add(input.command.join(' '));
    57	    if (Array.isArray(input?.parsed_cmd)) {
    58	        input.parsed_cmd.forEach((entry: any) => add(entry?.cmd));
    59	    }
    60	    return Array.from(candidates);
    61	}

   113	export function stringifyInspectableValue(value: unknown): string {
   114	    if (typeof value === 'string') {
   115	        const parsed = parseInspectableString(value);
   116	        if (parsed !== value) return stringifyInspectableValue(parsed);
   117	        return value.length > 0 ? value : '""';

   120	    return JSON.stringify(value, (_key, nested) => {
   121	        if (nested && typeof nested === 'object') {
   122	            if (seen.has(nested)) return '[Circular]';
   123	            seen.add(nested);
   124	        }
   125	        return nested;

   207	        command: previewCommand,
   208	        stdout: previewStdout,

    33	function unwrapShellCommand(command: string): string {
    34	    const match = command.match(/^(?:\/bin\/)?(?:ba|z)?sh\s+-l?c\s+([\s\S]+)$/);
    35	    if (!match) return command;
    36	    const inner = match[1].trim();
    37	    if (
    38	        (inner.startsWith('"') && inner.endsWith('"')) ||
    39	        (inner.startsWith("'") && inner.endsWith("'"))
    40	    ) {
    41	        return inner.slice(1, -1).trim();
    42	    }
    43	    return inner;
    44	}

    46	function getCommandCandidates(input: ToolCall['input']): string[] {
    47	    const candidates = new Set<string>();
    48	    const add = (value: unknown) => {
    49	        if (typeof value !== 'string') return;
    50	        const trimmed = value.trim();
    51	        if (!trimmed) return;
    52	        candidates.add(trimmed);
    53	        candidates.add(unwrapShellCommand(trimmed));
    54	    };
    55	    add(stringifyToolCommand(input?.command));
    56	    if (Array.isArray(input?.command)) add(input.command.join(' '));
    57	    if (Array.isArray(input?.parsed_cmd)) {
    58	        input.parsed_cmd.forEach((entry: any) => add(entry?.cmd));
    59	    }
    60	    return Array.from(candidates);
    61	}
