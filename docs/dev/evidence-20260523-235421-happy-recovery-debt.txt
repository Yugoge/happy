    37	  [ "${size:-0}" -lt "$MAX_LOG_BYTES" ] && return 0
    38	  for i in 5 4 3 2 1; do
    39	    [ -f "$LOG_FILE.$i" ] && mv -f "$LOG_FILE.$i" "$LOG_FILE.$((i+1))"

    41	  mv -f "$LOG_FILE" "$LOG_FILE.1"
